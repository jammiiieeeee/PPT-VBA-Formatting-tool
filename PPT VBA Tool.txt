' ==================================================
' PPT VBA Tools — Check & Flatten Presentations
' ==================================================
' Paste into VBA editor: Alt+F11 → Insert → Module
' Then Alt+F8 → run one of:
'   PPTTools_CheckSlides → inspect for issues (no changes)
'   PPTTools_Flatten     → flatten presentation (assumes clean)
' ==================================================

Option Explicit

' -------------------------------------------------------------------
' DEBUG MODE — set to True to see per-slide/per-shape diagnostic detail
' in the check log. Normal mode shows only actionable summary counts.
' -------------------------------------------------------------------
Private Const DEBUG_MODE As Boolean = False

' -------------------------------------------------------------------
' Data types — replace module-level mutable arrays.
' All state flows through AnalysisResult; no globals needed.
' -------------------------------------------------------------------

Private Type SlideData
    Index As String
    Text As String
    OrigTitle As String
    FinalTitle As String
    ShapeName As String
    MatchLength As Long
    Skip As Boolean
    SkipReason As String
    CheckedShapes As String
    DiscoveredTexts As String
End Type

Private Type AnalysisResult
    Slides() As SlideData
    TotalSlides As Long
    SlideWidth As Single
    SlideHeight As Single
End Type

Private Type TitleResult
    Index As String
    Text As String
    OrigTitle As String
    ShapeName As String
    MatchLength As Long
    Skip As Boolean
    SkipReason As String
    CheckedShapes As String
    DiscoveredTexts As String
End Type

' ===================================================================
' ENTRY POINTS  — these appear in Alt+F8
' ===================================================================

Public Sub PPTTools_CheckSlides()
    RunAllChecks
End Sub

Public Sub PPTTools_Flatten()
    FlattenPresentation
End Sub

' ===================================================================
' SHARED: CopySlideData
' Deep-copies a SlideData() array. Used for backup before
' AnalyseDuplicates so ProcessFixableChanges can revert on decline.
' ===================================================================

Private Function CopySlideData(ByRef src() As SlideData) As SlideData()
    Dim dst() As SlideData
    Dim lo As Long, hi As Long, i As Long
    lo = LBound(src): hi = UBound(src)
    ReDim dst(lo To hi)
    For i = lo To hi
        dst(i) = src(i)
    Next i
    CopySlideData = dst
End Function

' ===================================================================
' SHARED: PopulateTitleArrays
' Extracts titles for all slides using depth-based selection.
' Returns an AnalysisResult — no module-level state written.
' ===================================================================

Private Function PopulateTitleArrays(ByRef origPres As Presentation, _
                                     Optional ByVal captureDebug As Boolean = False) As AnalysisResult

    Dim result As AnalysisResult
    Dim regex As Object
    Dim slideNum As Long, objSlide As Slide

    result.SlideWidth = origPres.PageSetup.SlideWidth
    result.SlideHeight = origPres.PageSetup.SlideHeight
    result.TotalSlides = origPres.Slides.Count

    Set regex = CreateObject("VBScript.RegExp")
    With regex
        .Pattern = "^(\d+(?:[\.\(/\\\-_\??]\d+)*)\s*(.*)"
        .IgnoreCase = True
        .Global = True
    End With

    ReDim result.Slides(1 To result.TotalSlides)

    For slideNum = 1 To result.TotalSlides
        Set objSlide = origPres.Slides(slideNum)
        Dim titleResult As TitleResult
        titleResult = ExtractSlideTitle(objSlide, result.SlideHeight, regex, slideNum, captureDebug)

        With result.Slides(slideNum)
            .Skip = titleResult.Skip
            .SkipReason = titleResult.SkipReason
            .ShapeName = titleResult.ShapeName
            .MatchLength = titleResult.MatchLength
            .Index = titleResult.Index
            .Text = titleResult.Text
            .OrigTitle = titleResult.OrigTitle
            .FinalTitle = titleResult.OrigTitle
            .CheckedShapes = titleResult.CheckedShapes
            .DiscoveredTexts = titleResult.DiscoveredTexts
        End With
    Next slideNum

    PopulateTitleArrays = result
End Function

' ===================================================================
' SHARED: CountSeparators
' Counts separator characters (. ( / \ - _ ?) in an index string.
' ===================================================================

Private Function CountSeparators(ByVal s As String) As Long
    Dim r As String: r = s
    r = Replace(r, ".", "")
    r = Replace(r, "(", "")
    r = Replace(r, "-", "")
    r = Replace(r, "_", "")
    r = Replace(r, "/", "")
    r = Replace(r, "\", "")
    r = Replace(r, "?", "")
    CountSeparators = Len(s) - Len(r)
End Function

' ===================================================================
' SHARED: ExtractSlideTitle
' Returns TitleResult for one slide: guardrails, override, depth
' selection, tie-breaker, and no-match fallback.
' ===================================================================

Private Function ExtractSlideTitle(ByVal sld As Slide, ByVal sldHeight As Single, _
                                   ByVal regex As Object, ByVal slideNum As Long, _
                                   Optional ByVal captureDebug As Boolean = False) As TitleResult

    Dim result As TitleResult
    Dim objShape As Shape

    result.Skip = False
    result.SkipReason = ""
    result.ShapeName = ""
    result.MatchLength = 0
    result.CheckedShapes = ""
    result.DiscoveredTexts = ""

    ' Guardrail 1: Keyword match
    For Each objShape In sld.Shapes
        If objShape.Visible = msoTrue And objShape.HasTextFrame And objShape.TextFrame.HasText Then
            Dim lowerText As String
            lowerText = LCase(Trim(objShape.TextFrame.TextRange.Text))
            If InStr(lowerText, "thank you") > 0 Or _
               (InStr(lowerText, "thanks") > 0 And InStr(lowerText, "thanksgiving") = 0) Or _
               InStr(lowerText, "??") > 0 Or _
               InStr(lowerText, "q&a") > 0 Or _
               InStr(lowerText, "q & a") > 0 Then
                result.Skip = True
                result.SkipReason = "Keyword match"
                ExtractSlideTitle = result
                Exit Function
            End If
        End If
    Next objShape

    ' Guardrail 2: Non-standard slide name
    If sld.Name <> "Slide" & slideNum And Left(sld.Name, 5) <> "Slide" Then
        result.Skip = True
        result.SkipReason = "Non-standard slide name"
        ExtractSlideTitle = result
        Exit Function
    End If

    ' Override: priority keywords bypass depth-based selection
    For Each objShape In sld.Shapes
        If objShape.Visible = msoTrue And objShape.HasTextFrame And objShape.TextFrame.HasText Then
            Dim overrideText As String
            overrideText = LCase(Trim(objShape.TextFrame.TextRange.Text))
            If InStr(overrideText, "course objective") > 0 Or _
               InStr(overrideText, "learning outcomes") > 0 Or _
               InStr(overrideText, "table of content") > 0 Or _
                InStr(overrideText, ChrW(&H5B66) & ChrW(&H4E60) & ChrW(&H76EE) & ChrW(&H6807)) > 0 Or _
                InStr(overrideText, ChrW(&H5B66) & ChrW(&H4E60) & ChrW(&H6210) & ChrW(&H679C)) > 0 Or _
                InStr(overrideText, ChrW(&H5185) & ChrW(&H5BB9) & ChrW(&H76EE) & ChrW(&H5F55)) > 0 Or _
                InStr(overrideText, ChrW(&H57F9) & ChrW(&H8BAD) & ChrW(&H76EE) & ChrW(&H6807)) > 0 Or _
                InStr(overrideText, ChrW(&H76EE) & ChrW(&H5F55)) > 0 Or _
                InStr(overrideText, ChrW(&H5B66) & ChrW(&H4E60) & ChrW(&H6536) & ChrW(&H83B7)) > 0 Then

                If captureDebug Then
                    result.CheckedShapes = "[" & objShape.Name & "] "
                    result.DiscoveredTexts = "    -> [OVERRIDE TITLE] '" & Trim(Left(objShape.TextFrame.TextRange.Text, 25)) & "...'"
                End If

                result.ShapeName = objShape.Name
                result.MatchLength = 0
                result.Index = ""
                result.Text = Trim(objShape.TextFrame.TextRange.Text)
                result.OrigTitle = result.Text
                ExtractSlideTitle = result
                Exit Function
            End If
        End If
    Next objShape

    ' Depth-based title selection
    Dim deepestIndex As String: deepestIndex = ""
    Dim deepestText As String: deepestText = ""
    Dim deepestMatchLen As Long: deepestMatchLen = 0
    Dim deepestShapeName As String: deepestShapeName = ""
    Dim maxDots As Long: maxDots = -1
    Dim matches As Object

    For Each objShape In sld.Shapes
        If objShape.Visible = msoTrue Then
            If captureDebug Then
                result.CheckedShapes = result.CheckedShapes & "[" & objShape.Name & "] "
            End If

            If objShape.HasTextFrame And objShape.TextFrame.HasText Then
                Dim fullText As String
                fullText = objShape.TextFrame.TextRange.Text
                fullText = Replace(fullText, vbCrLf, vbCr)
                fullText = Replace(fullText, vbLf, vbCr)

                Dim textBlocks() As String
                textBlocks = Split(fullText, vbCr)

                Dim blockIdx As Long
                For blockIdx = 0 To UBound(textBlocks)
                    Dim currentBlockText As String
                    currentBlockText = Trim(textBlocks(blockIdx))

                    If Len(currentBlockText) > 0 Then
                        Dim croppedTxt As String: croppedTxt = Left(currentBlockText, 25)

                        If objShape.Top < (sldHeight * 0.5) Then
                            If regex.Test(currentBlockText) Then
                                Set matches = regex.Execute(currentBlockText)
                                Dim matchObj As Object
                                For Each matchObj In matches
                                    Dim currentIndex As String, currentText As String
                                    currentIndex = matchObj.SubMatches(0)
                                    currentText = matchObj.SubMatches(1)

                                    Dim currentDots As Long
                                    currentDots = CountSeparators(currentIndex)

                                    If captureDebug Then
                                        result.DiscoveredTexts = result.DiscoveredTexts & _
                                            vbCrLf & "    -> [MATCHED BLOCK] '" & croppedTxt & "...' (Shape: " & objShape.Name & ", Index: '" & currentIndex & "', Separators: " & currentDots & ", Top: " & Round(objShape.Top, 1) & ")"
                                    End If

                                    If currentDots > maxDots Then
                                        maxDots = currentDots
                                        deepestIndex = currentIndex
                                        deepestText = currentText
                                        deepestMatchLen = matchObj.Length
                                        deepestShapeName = objShape.Name
                                    End If
                                Next matchObj
                            Else
                                If captureDebug Then
                                    result.DiscoveredTexts = result.DiscoveredTexts & _
                                        vbCrLf & "    -> [REJECTED BY REGEX] '" & croppedTxt & "...' (Shape: " & objShape.Name & ")"
                                End If
                            End If
                        Else
                            If captureDebug Then
                                result.DiscoveredTexts = result.DiscoveredTexts & _
                                    vbCrLf & "    -> [REJECTED BY HEIGHT GATE] '" & croppedTxt & "...' (Shape: " & objShape.Name & ", Top: " & Round(objShape.Top, 1) & ")"
                            End If
                        End If
                    End If
                Next blockIdx
            End If
        End If
    Next objShape

    ' Tie-breaker detection
    If deepestIndex <> "" Then
        Dim distinctMatchCount As Long: distinctMatchCount = 0
        Dim tiedShapeNames As String: tiedShapeNames = ""

        For Each objShape In sld.Shapes
            If objShape.Visible = msoTrue And objShape.HasTextFrame Then
                If objShape.TextFrame.HasText And objShape.Top < (sldHeight * 0.5) Then
                    Dim scanText As String
                    scanText = objShape.TextFrame.TextRange.Text
                    scanText = Replace(scanText, vbCrLf, vbCr)
                    scanText = Replace(scanText, vbLf, vbCr)

                    Dim scanBlocks() As String
                    scanBlocks = Split(scanText, vbCr)

                    Dim sIdx As Long
                    For sIdx = 0 To UBound(scanBlocks)
                        Dim checkBlock As String: checkBlock = Trim(scanBlocks(sIdx))
                        If regex.Test(checkBlock) Then
                            Set matches = regex.Execute(checkBlock)
                            Dim testIdx As String
                            testIdx = matches(0).SubMatches(0)

                            If CountSeparators(testIdx) = maxDots Then
                                If InStr(tiedShapeNames, "|" & objShape.Name & "|") = 0 Then
                                    tiedShapeNames = tiedShapeNames & "|" & objShape.Name & "|"
                                    distinctMatchCount = distinctMatchCount + 1
                                End If
                            End If
                        End If
                    Next sIdx
                End If
            End If
        Next objShape

        If distinctMatchCount > 1 Then
            result.Skip = True
            result.SkipReason = "Multiple shapes tie at separator depth " & maxDots
            result.Index = ""
            result.Text = ""
            result.ShapeName = ""
            result.MatchLength = 0
            result.OrigTitle = ""
            ExtractSlideTitle = result
            Exit Function
        End If

        result.Index = deepestIndex
        result.Text = Trim(deepestText)
        result.ShapeName = deepestShapeName
        result.MatchLength = deepestMatchLen

        If Len(result.Text) > 0 Then
            result.OrigTitle = deepestIndex & " " & result.Text
        Else
            result.OrigTitle = deepestIndex
        End If
    Else
        result.Skip = True
        result.SkipReason = "No valid title candidate found"
        result.Index = ""
        result.Text = ""
        result.OrigTitle = ""
    End If

    ExtractSlideTitle = result
End Function

' ===================================================================
' SHARED: CountCharacters
' Counts total non-whitespace characters, Latin (A-Z a-z), and
' non-Latin characters in a string.
' ===================================================================

Private Sub CountCharacters(ByVal text As String, ByRef total As Long, _
                            ByRef latin As Long, ByRef nonLatin As Long)
    total = 0: latin = 0: nonLatin = 0
    Dim i As Long, c As String, code As Long
    For i = 1 To Len(text)
        c = Mid(text, i, 1)
        If c <> " " And c <> vbCr And c <> vbLf And c <> vbTab Then
            total = total + 1
            code = AscW(c)
            If (code >= 65 And code <= 90) Or (code >= 97 And code <= 122) Then
                latin = latin + 1
            Else
                nonLatin = nonLatin + 1
            End If
        End If
    Next i
End Sub

' ===================================================================
' SHARED: DetectLanguage
' Returns "ZH" (Chinese), "EN" (English), "MIXED", or "" (empty)
' Based on CJK vs Latin character ratios.
' ===================================================================

Private Function DetectLanguage(ByVal text As String) As String
    If Len(Trim(text)) = 0 Then
        DetectLanguage = ""
        Exit Function
    End If

    Dim total As Long, latin As Long, nonLatin As Long
    CountCharacters text, total, latin, nonLatin

    If total = 0 Then
        DetectLanguage = ""
    ElseIf (nonLatin / total) >= 0.5 Then
        DetectLanguage = "ZH"
    ElseIf (latin / total) >= 0.5 Then
        DetectLanguage = "EN"
    Else
        DetectLanguage = "MIXED"
    End If
End Function

' ===================================================================
' SHARED: GetNotesText
' Returns notes text from a slide, or "" if missing/empty.
' Uses NotesPage (SlideRange) — works across PPT versions that
' don't expose NotesSlide on the Slide object.
' ===================================================================

Private Function GetNotesText(ByRef sld As Object) As String
    Dim np As Object
    On Error Resume Next
    Set np = sld.NotesPage
    If Not np Is Nothing Then
        Dim ph As Object
        Set ph = np.Shapes.Placeholders(2)
        If Not ph Is Nothing Then
            If ph.HasTextFrame Then
                If ph.TextFrame.HasText Then
                    Dim txt As String
                    txt = Trim(ph.TextFrame.TextRange.Text)
                    If Len(txt) > 0 Then
                        GetNotesText = txt
                    End If
                End If
            End If
        End If
    End If
    On Error GoTo 0
End Function

' ===================================================================
' SHARED: WriteLog
' Writes content to a timestamped log file on the Desktop.
' ===================================================================

Private Function WriteLog(ByVal content As String, ByVal saveDir As String) As String
    Dim baseName As String, reportPath As String
    baseName = saveDir & "\Articulate_Title_Update_Log_" & Format(Now, "yyyymmdd_hhmmss")
    reportPath = baseName & ".txt"
    ' Avoid overwriting if two runs land in the same second
    Dim fileCounter As Long: fileCounter = 2
    Do While Dir(reportPath) <> ""
        reportPath = baseName & "_" & fileCounter & ".txt"
        fileCounter = fileCounter + 1
    Loop
    Dim fileNum As Integer
    fileNum = FreeFile
    On Error Resume Next
    Open reportPath For Output As #fileNum
    If Err.Number <> 0 Then
        Err.Clear
        ' Fall back to Desktop if the original path fails (e.g. OneDrive/SharePoint)
        baseName = CreateObject("WScript.Shell").SpecialFolders("Desktop") & _
                   "\Articulate_Title_Update_Log_" & Format(Now, "yyyymmdd_hhmmss")
        reportPath = baseName & ".txt"
        fileCounter = 2
        Do While Dir(reportPath) <> ""
            reportPath = baseName & "_" & fileCounter & ".txt"
            fileCounter = fileCounter + 1
        Loop
        fileNum = FreeFile
        Open reportPath For Output As #fileNum
    End If
    On Error GoTo 0
    Print #fileNum, content
    Close #fileNum
    WriteLog = reportPath
End Function

' ===================================================================
' CHECKING ACTION: RunAllChecks
' Runs full analysis: titles, errors, notes audit, translation gap.
' Writes comprehensive log. Does NOT modify the presentation.
' ===================================================================

Private Sub RunAllChecks()
    Dim origPres As Presentation
    Set origPres = PowerPoint.ActivePresentation
    If origPres.Path = "" Then
        MsgBox "Please save your presentation before running checks.", vbCritical, "Save Required"
        Exit Sub
    End If
    If origPres.Slides.Count = 0 Then
        MsgBox "Presentation has no slides.", vbExclamation, "Nothing to Check"
        Exit Sub
    End If

    ' Phase 1: Extract titles — returns local AnalysisResult, no globals
    Dim result As AnalysisResult
    result = PopulateTitleArrays(origPres, captureDebug:=True)

    ' Compute ppt-level language from all non-skipped title texts
    Dim slideNum As Long
    Dim fullTitleText As String, pptLang As String
    fullTitleText = ""
    For slideNum = 1 To result.TotalSlides
        If Not result.Slides(slideNum).Skip And result.Slides(slideNum).Text <> "" Then
            fullTitleText = fullTitleText & " " & result.Slides(slideNum).Text
        End If
    Next slideNum
    pptLang = DetectLanguage(fullTitleText)
    If pptLang = "" Or pptLang = "MIXED" Then
        Dim t As Long, zhCount As Long, enCount As Long
        zhCount = 0: enCount = 0
        For t = 1 To result.TotalSlides
            If Not result.Slides(t).Skip And result.Slides(t).Text <> "" Then
                Dim lang As String
                lang = DetectLanguage(result.Slides(t).Text)
                If lang = "ZH" Then zhCount = zhCount + 1
                If lang = "EN" Then enCount = enCount + 1
            End If
        Next t
        If zhCount > enCount Then
            pptLang = "ZH"
        ElseIf enCount > zhCount Then
            pptLang = "EN"
        Else
            MsgBox "Cannot determine presentation language." & vbCrLf & vbCrLf & _
                   "ZH titles: " & zhCount & ", EN titles: " & enCount, vbCritical, "Language Detection Failed"
            Exit Sub
        End If
    End If

    ' Phase 2: Error analysis — backup before AnalyseDuplicates modifies slides
    Dim reportDetails As String, fixableCount As Long, unfixableCount As Long
    Dim fixableSlides As String, nonFixableSlides As String
    Dim backupSlides() As SlideData
    backupSlides = CopySlideData(result.Slides)

    AnalyseDuplicates result.Slides, result.TotalSlides, reportDetails, fixableCount, unfixableCount, fixableSlides, nonFixableSlides

    ' Phase 3: Prompt user to apply fixable changes (grouped by consecutive runs)
    ' Returns True if any slides were reverted (arrays rolled back to backup).
    Dim anyReverted As Boolean
    anyReverted = ProcessFixableChanges(origPres, result.Slides, backupSlides, result.TotalSlides)

    ' Phase 4: Re-run analysis if slides were reverted so the log reflects
    ' the true remaining state (unfixed issues get fresh suffix/renumbering).
If anyReverted Then
        reportDetails = ""
        fixableCount = 0
        unfixableCount = 0
        AnalyseDuplicates result.Slides, result.TotalSlides, reportDetails, fixableCount, unfixableCount, fixableSlides, nonFixableSlides
    End If

    ' Phase 5: Notes audit
    Dim emptyNotesCount As Long, emptyNotesSlides As String, notesLog As String
    AuditNotes origPres, result.TotalSlides, emptyNotesCount, emptyNotesSlides, notesLog

    ' Phase 6: Translation gap check
    Dim shapeGapCount As Long, notesGapCount As Long, visGapSlides As Long
    Dim visShapesChecked As Long, transSummary As String
    Dim transGapNoteSlides As String, transGapShapeSlides As String
    CheckTranslationGaps origPres, result.Slides, result.TotalSlides, pptLang, shapeGapCount, notesGapCount, visGapSlides, visShapesChecked, transSummary, transGapNoteSlides, transGapShapeSlides

    ' Phase 7: Skipped slides summary
    Dim skippedSummary As String
    BuildSkippedSummary result.Slides, result.TotalSlides, skippedSummary

    ' Phase 8: Debug inspection matrix (only in DEBUG_MODE)
    Dim matrixLog As String
    If DEBUG_MODE Then
        BuildDebugMatrix result.Slides, result.TotalSlides, matrixLog
    Else
        matrixLog = ""
    End If

    ' Phase 9: Build comprehensive summary
    Dim totalSkipped As Long, totalProcessed As Long, totalWithNotes As Long
    totalSkipped = 0
    For slideNum = 1 To result.TotalSlides
        If result.Slides(slideNum).Skip Then totalSkipped = totalSkipped + 1
    Next slideNum
    totalProcessed = result.TotalSlides - totalSkipped
    totalWithNotes = result.TotalSlides - emptyNotesCount

    Dim reportSummary As String
    reportSummary = "==================================================" & vbCrLf
    reportSummary = reportSummary & "METRIC OVERVIEW" & vbCrLf
    reportSummary = reportSummary & "==================================================" & vbCrLf
    reportSummary = reportSummary & "TIMESTAMP: " & Now & vbCrLf
    reportSummary = reportSummary & "TOTAL SLIDES: " & result.TotalSlides & vbCrLf
    reportSummary = reportSummary & "PROCESSED (non-skipped): " & totalProcessed & vbCrLf
    reportSummary = reportSummary & "SKIPPED: " & totalSkipped & vbCrLf
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "--- LANGUAGE ---" & vbCrLf
    reportSummary = reportSummary & "  Presentation language: " & pptLang & vbCrLf
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "--- TITLE ISSUES ---" & vbCrLf
    reportSummary = reportSummary & "  Fixable:     " & fixableCount
    If fixableCount > 0 Then
        reportSummary = reportSummary & "  (contiguous duplicate titles — auto-suffixed during flatten)"
    End If
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "  Non-fixable: " & unfixableCount
    If unfixableCount > 0 Then
        reportSummary = reportSummary & "  (same title on non-consecutive slides — manual rename needed)"
    End If
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "--- NOTES ---" & vbCrLf
    reportSummary = reportSummary & "  Slides with notes: " & totalWithNotes & vbCrLf
    reportSummary = reportSummary & "  Empty notes:       " & emptyNotesCount & vbCrLf
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "--- TRANSLATION GAPS ---" & vbCrLf
    reportSummary = reportSummary & "  Shapes checked:   " & visShapesChecked & vbCrLf
    reportSummary = reportSummary & "  Shape gaps:       " & shapeGapCount & " (slides: " & visGapSlides & ")" & vbCrLf
    reportSummary = reportSummary & "  Notes gaps:       " & notesGapCount & vbCrLf
    reportSummary = reportSummary & vbCrLf
    reportSummary = reportSummary & "==================================================" & vbCrLf & vbCrLf

    ' Build NEXT STEPS with slide numbers per problem type — placed at top
    Dim nextSteps As String
    nextSteps = "NEXT STEPS" & vbCrLf
    Dim hasIssues As Boolean: hasIssues = False
    If fixableCount > 0 Or unfixableCount > 0 Or emptyNotesCount > 0 Or shapeGapCount > 0 Or notesGapCount > 0 Then hasIssues = True
    If hasIssues Then
        If fixableCount > 0 Then
            nextSteps = nextSteps & "  [FIXABLE] " & fixableCount & " contiguous duplicate title run(s) — slides: " & fixableSlides & vbCrLf
            nextSteps = nextSteps & "    → Will be auto-suffixed during flatten (e.g., \"Title (1/3)\", \"Title (2/3)\", \"Title (3/3)\")" & vbCrLf
        End If
        If unfixableCount > 0 Then
            nextSteps = nextSteps & "  [NON-FIXABLE] " & unfixableCount & " non-contiguous duplicate title(s) — slides: " & nonFixableSlides & vbCrLf
            nextSteps = nextSteps & "    → Rename manually before flattening (same title on non-adjacent slides)" & vbCrLf
        End If
        If emptyNotesCount > 0 Then
            nextSteps = nextSteps & "  [EMPTY NOTES] " & emptyNotesCount & " slide(s) — slides: " & emptyNotesSlides & vbCrLf
            nextSteps = nextSteps & "    → Add speaker notes before flattening" & vbCrLf
        End If
        If shapeGapCount > 0 Then
            nextSteps = nextSteps & "  [SHAPE TRANSLATION GAP] " & shapeGapCount & " shape(s) — slides: " & transGapShapeSlides & vbCrLf
            nextSteps = nextSteps & "    → Shape text language mismatches presentation language" & vbCrLf
        End If
        If notesGapCount > 0 Then
            nextSteps = nextSteps & "  [NOTES TRANSLATION GAP] " & notesGapCount & " slide(s) — slides: " & transGapNoteSlides & vbCrLf
            nextSteps = nextSteps & "    → Notes language mismatches presentation language" & vbCrLf
        End If
        nextSteps = nextSteps & vbCrLf & "  After fixing, re-run PPTTools_CheckSlides to verify, then run PPTTools_Flatten."
    Else
        nextSteps = nextSteps & "  No issues found. Ready to run PPTTools_Flatten."
    End If
    nextSteps = nextSteps & vbCrLf & vbCrLf
    nextSteps = nextSteps & "==================================================" & vbCrLf & vbCrLf

    ' Phase 11: Write report (post-fix state) — NEXT STEPS at top, then details
    Dim fullReport As String
    fullReport = reportSummary & nextSteps & reportDetails & notesLog & transSummary & skippedSummary & matrixLog
    Dim logPath As String
    logPath = WriteLog(fullReport, origPres.Path)

    Dim msg As String
    msg = "Check complete." & vbCrLf & vbCrLf
    If fixableCount > 0 Then msg = msg & "Title issues (fixable): " & fixableCount & vbCrLf
    If unfixableCount > 0 Then msg = msg & "Title issues (needs manual fix): " & unfixableCount & vbCrLf
    If emptyNotesCount > 0 Then msg = msg & "Slides with no notes: " & emptyNotesCount & vbCrLf
    If shapeGapCount > 0 Then msg = msg & "Shape text gaps: " & shapeGapCount & vbCrLf
    If notesGapCount > 0 Then msg = msg & "Notes gaps: " & notesGapCount & vbCrLf
    msg = msg & vbCrLf & "Presentation language: " & pptLang

    If MsgBox(msg & vbCrLf & vbCrLf & "Open the log file now?", vbYesNo + vbQuestion, "Check Complete") = vbYes Then
        Shell "notepad.exe """ & logPath & """", vbNormalFocus
    End If
End Sub

Private Sub AnalyseDuplicates(ByRef slides() As SlideData, _
                              ByVal totalSlides As Long, _
                              ByRef reportDetails As String, _
                              ByRef fixableCount As Long, _
                              ByRef unfixableCount As Long, _
                              ByRef fixableSlides As String, _
                              ByRef nonFixableSlides As String)
    Dim slideNum As Long
    reportDetails = ""
    fixableCount = 0: unfixableCount = 0
    fixableSlides = "": nonFixableSlides = ""

    ' Case 1: Contiguous duplicate detection
    Dim runStart As Long, runEnd As Long, runIdx As Long, suffix As String, j As Long
    slideNum = 1
    Do While slideNum <= totalSlides
        If slides(slideNum).Text <> "" And Not slides(slideNum).Skip Then
            runStart = slideNum: runEnd = slideNum
            Do While (runEnd + 1 <= totalSlides)
                If slides(runEnd + 1).Text = slides(runStart).Text And Not slides(runEnd + 1).Skip Then
                    runEnd = runEnd + 1
                Else
                    Exit Do
                End If
            Loop

            Dim totalInRun As Long: totalInRun = (runEnd - runStart) + 1
            If totalInRun > 1 Then
                fixableCount = fixableCount + 1
                ' Track fixable slide numbers
                Dim fs As String: fs = ""
                For runIdx = runStart To runEnd
                    If fs <> "" Then fs = fs & ", "
                    fs = fs & runIdx
                Next runIdx
                If fixableSlides <> "" Then fixableSlides = fixableSlides & ", "
                fixableSlides = fixableSlides & fs
                If DEBUG_MODE Then
                    reportDetails = reportDetails & "-> [FIXABLE]:" & vbCrLf & _
                                    "    Contiguous duplicate titles on Slides " & runStart & "-" & runEnd & " matching '" & slides(slideNum).Text & "'. Suffix markers added." & vbCrLf & vbCrLf
                End If

                For runIdx = runStart To runEnd
                    suffix = " (" & (runIdx - runStart + 1) & "/" & totalInRun & ")"
                    slides(runIdx).Text = slides(runIdx).Text & suffix
                    If slides(runIdx).Index <> "" Then
                        If Len(slides(runIdx).Text) > 0 Then
                            slides(runIdx).FinalTitle = slides(runIdx).Index & " " & slides(runIdx).Text
                        Else
                            slides(runIdx).FinalTitle = slides(runIdx).Index
                        End If
                    Else
                        slides(runIdx).FinalTitle = slides(runIdx).Text
                    End If
                Next runIdx
                slideNum = runEnd
            Else
                Dim foundNonContig As Boolean: foundNonContig = False
                For j = 1 To totalSlides
                    If j < runStart Or j > runEnd Then
                        If slides(j).Text = slides(slideNum).Text And slides(slideNum).Text <> "" And Not slides(j).Skip Then
                            foundNonContig = True
                            Exit For
                        End If
                    End If
                Next j

                If foundNonContig Then
                    Dim alreadyReported As Boolean: alreadyReported = False
                    Dim prev As Long
                    For prev = 1 To slideNum - 1
                        If slides(prev).Text = slides(slideNum).Text And Not slides(prev).Skip Then
                            alreadyReported = True
                            Exit For
                        End If
                    Next prev

                    If Not alreadyReported Then
                        Dim allSlides As String: allSlides = ""
                        Dim slideIdx As Long
                        For slideIdx = 1 To totalSlides
                            If slides(slideIdx).Text = slides(slideNum).Text And Not slides(slideIdx).Skip Then
                                If allSlides <> "" Then allSlides = allSlides & ", "
                                allSlides = allSlides & slideIdx
                            End If
                        Next slideIdx

                        unfixableCount = unfixableCount + 1
                        ' Track non-fixable slide numbers
                        If nonFixableSlides <> "" Then nonFixableSlides = nonFixableSlides & ", "
                        nonFixableSlides = nonFixableSlides & allSlides
                        If DEBUG_MODE Then
                            reportDetails = reportDetails & "-> [NON-FIXABLE]:" & vbCrLf & _
                                            "    Slides " & allSlides & " share value '" & slides(slideNum).Text & "'." & vbCrLf & vbCrLf
                        End If
                    End If
                End If
            End If
        End If
        slideNum = slideNum + 1
    Loop

    ' Case 2: Conflicting numerical index detection
    Dim lastLevelNum As Long, k As Long
    slideNum = 1
    Do While slideNum < totalSlides
        If slides(slideNum).Index <> "" And slides(slideNum + 1).Index <> "" And Not slides(slideNum).Skip And Not slides(slideNum + 1).Skip Then
            If slides(slideNum).Index = slides(slideNum + 1).Index Then
                fixableCount = fixableCount + 1

                Dim originConflictStart As Long: originConflictStart = slideNum
                Dim originConflictEnd As Long: originConflictEnd = slideNum + 1

                ' Use regex to safely extract last numeric component.
                ' This handles mixed separators (e.g. "1.2/3") that would
                ' crash Split + CLng on a single delimiter character.
                Dim idxRegex As Object
                Set idxRegex = CreateObject("VBScript.RegExp")
                idxRegex.Pattern = "^(.*?)(\d+)$"
                Dim idxMatch As Object
                Set idxMatch = idxRegex.Execute(slides(slideNum).Index)

                If idxMatch.Count = 0 Then GoTo SkipIndexFix

                Dim baseUpperSection As String: baseUpperSection = idxMatch(0).SubMatches(0)
                lastLevelNum = CLng(idxMatch(0).SubMatches(1))
                Dim expectedDepth As Long: expectedDepth = CountSeparators(slides(slideNum).Index) + 1

                Dim totalAffectedEnd As Long: totalAffectedEnd = slideNum + 1
                k = slideNum + 1
                Do While k <= totalSlides
                    If slides(k).Index <> "" And Not slides(k).Skip Then
                        If Left(slides(k).Index, Len(baseUpperSection)) = baseUpperSection Then
                            If CountSeparators(slides(k).Index) + 1 = expectedDepth Then
                                totalAffectedEnd = k
                            Else
                                Exit Do
                            End If
                        Else
                            Exit Do
                        End If
                    Else
                        Exit Do
                    End If
                    k = k + 1
                Loop

                ' Track fixable slide numbers (Case 2: index conflict)
                Dim fs2 As String: fs2 = ""
                For k = originConflictStart To totalAffectedEnd
                    If fs2 <> "" Then fs2 = fs2 & ", "
                    fs2 = fs2 & k
                Next k
                If fixableSlides <> "" Then fixableSlides = fixableSlides & ", "
                fixableSlides = fixableSlides & fs2
                If DEBUG_MODE Then
                    reportDetails = reportDetails & "-> [FIXABLE]:" & vbCrLf & _
                                    "    Index conflict '" & slides(slideNum).Index & "' on Slides " & originConflictStart & "-" & originConflictEnd & "." & vbCrLf & _
                                    "    Cascading impact: Shifted layouts from Slides " & originConflictStart & " to " & totalAffectedEnd & "." & vbCrLf & vbCrLf
                End If

                k = slideNum + 1
                Do While k <= totalSlides
                    If slides(k).Index <> "" And Not slides(k).Skip Then
                        If Left(slides(k).Index, Len(baseUpperSection)) = baseUpperSection Then
                            If CountSeparators(slides(k).Index) + 1 = expectedDepth Then
                                lastLevelNum = lastLevelNum + 1
                                slides(k).Index = baseUpperSection & lastLevelNum
                                If Len(slides(k).Text) > 0 Then
                                    slides(k).FinalTitle = slides(k).Index & " " & slides(k).Text
                                Else
                                    slides(k).FinalTitle = slides(k).Index
                                End If
                            Else
                                Exit Do
                            End If
                        Else
                            Exit Do
                        End If
                    Else
                        Exit Do
                    End If
                    k = k + 1
                Loop
                slideNum = k - 1
SkipIndexFix:
            End If
        End If
        slideNum = slideNum + 1
    Loop
End Sub

Private Sub AuditNotes(ByRef origPres As Presentation, _
                       ByVal totalSlides As Long, _
                       ByRef emptyNotesCount As Long, _
                       ByRef emptyNotesSlides As String, _
                       ByRef notesLog As String)
    Dim slideNum As Long, sld As Slide
    emptyNotesSlides = ""
    emptyNotesCount = 0

    For slideNum = 1 To totalSlides
        Set sld = origPres.Slides(slideNum)
        Dim notesTxt As String
        notesTxt = GetNotesText(sld)
        If notesTxt = "" Then
            emptyNotesCount = emptyNotesCount + 1
            If emptyNotesSlides <> "" Then emptyNotesSlides = emptyNotesSlides & ", "
            emptyNotesSlides = emptyNotesSlides & slideNum
        End If
    Next slideNum

    notesLog = "NOTES AUDIT" & vbCrLf & _
               "  Slides with no notes: " & emptyNotesCount
    If emptyNotesCount > 0 Then
        notesLog = notesLog & " (Slides " & emptyNotesSlides & ")"
    End If
    notesLog = notesLog & vbCrLf

    ' Empty notes debug — only show anomalies (NotesPage inaccessible or unexpected shape state)
    If emptyNotesCount > 0 Then
        Dim hasAnomaly As Boolean: hasAnomaly = False
        Dim anomalyLog As String: anomalyLog = ""
        For slideNum = 1 To totalSlides
            Set sld = origPres.Slides(slideNum)
            Dim notesDbgTxt As String
            notesDbgTxt = GetNotesText(sld)
            If notesDbgTxt = "" Then
                Dim npDbg As Object
                On Error Resume Next
                Set npDbg = sld.NotesPage
                If npDbg Is Nothing Then
                    hasAnomaly = True
                    anomalyLog = anomalyLog & "  Slide " & slideNum & ": NotesPage NOT ACCESSIBLE (Err=" & Err.Number & ")" & vbCrLf
                Else
                    ' Check for unexpected states: shapes with text frames but empty/missing text
                    Dim shpDbg As Object
                    For Each shpDbg In npDbg.Shapes
                        If shpDbg.HasTextFrame Then
                            If shpDbg.TextFrame.HasText Then
                                Dim shpTxt As String
                                shpTxt = Trim(shpDbg.TextFrame.TextRange.Text)
                                If Len(shpTxt) > 0 Then
                                    ' Shape has text but GetNotesText returned empty — anomaly
                                    hasAnomaly = True
                                    anomalyLog = anomalyLog & "  Slide " & slideNum & ": Shape '" & shpDbg.Name & "' has text but notes appear empty: """ & Trim(Left(shpTxt, 80)) & """" & vbCrLf
                                End If
                            End If
                        End If
                    Next shpDbg
                End If
                On Error GoTo 0
            End If
        Next slideNum
        If hasAnomaly Then
            notesLog = notesLog & "--- EMPTY NOTES ANOMALIES ---" & vbCrLf & anomalyLog
        End If
    End If
End Sub

Private Sub CheckTranslationGaps(ByRef origPres As Presentation, _
                                 ByRef slides() As SlideData, _
                                 ByVal totalSlides As Long, _
                                 ByVal pptLang As String, _
                                 ByRef shapeGapCount As Long, _
                                 ByRef notesGapCount As Long, _
                                 ByRef visGapSlides As Long, _
                                 ByRef visShapesChecked As Long, _
                                 ByRef transSummary As String, _
                                 ByRef transGapNoteSlides As String, _
                                 ByRef transGapShapeSlides As String)
    Dim slideNum As Long, sld As Slide, objShape As Shape
    shapeGapCount = 0: notesGapCount = 0
    visGapSlides = 0: visShapesChecked = 0
    Dim transGapLog As String: transGapLog = ""
    Dim gapNoteSlides As String: gapNoteSlides = ""
    Dim gapShapeSlides As String: gapShapeSlides = ""

    For slideNum = 1 To totalSlides
        Set sld = origPres.Slides(slideNum)
        Dim slideGapLog As String: slideGapLog = ""
        Dim hasGap As Boolean: hasGap = False

        ' Regex to identify technical fragments (measurements, tolerances, ranges)
        ' Matches: pure numbers, numbers with units, ranges (X to Y, X ± Y), isolated symbols
        Dim techFragRegex As Object
        Set techFragRegex = CreateObject("VBScript.RegExp")
        techFragRegex.Pattern = "^[\d\s\.\,\±\>\<\=\-\+\/\%\(\)]+$|^[\d\s\.\,\±\>\<\=\-\+\/\%\(\)]*(mm|cm|in|ft|yd|m|km|µm|nm|deg|°|rad|tol|min|max|typ|ref|to)[\d\s\.\,\±\>\<\=\-\+\/\%\(\)]*$"
        techFragRegex.IgnoreCase = True

        ' Visible text shape scan (non-skipped slides only)
        If Not slides(slideNum).Skip Then
            Dim visGapLog As String, visSlideGaps As Long
            visGapLog = ""
            visSlideGaps = 0

            For Each objShape In sld.Shapes
                If objShape.Visible = msoTrue And objShape.HasTextFrame And objShape.TextFrame.HasText Then
                    Dim visText As String
                    visText = Trim(objShape.TextFrame.TextRange.Text)
                    If Len(visText) > 0 Then
                        Dim visNW As Long, visLatin As Long, visNonLatin As Long
                        CountCharacters visText, visNW, visLatin, visNonLatin
                        If visNW >= 5 Then
                            visShapesChecked = visShapesChecked + 1
                            If slides(slideNum).OrigTitle <> "" And StrComp(visText, Trim(slides(slideNum).OrigTitle), vbTextCompare) = 0 Then GoTo SkipVisShape
                            Dim visPctLatin As Double, visPctNonLatin As Double
                            visPctLatin = visLatin / visNW
                            visPctNonLatin = visNonLatin / visNW
                            Dim visIsGap As Boolean
                            visIsGap = False
                            If pptLang = "ZH" And visPctLatin >= 0.9 Then visIsGap = True
                            If pptLang = "EN" And visPctNonLatin >= 0.9 Then visIsGap = True
                            If visIsGap Then
                                visSlideGaps = visSlideGaps + 1
                                shapeGapCount = shapeGapCount + 1
                                hasGap = True
                                If DEBUG_MODE Then
                                    visGapLog = visGapLog & "      SHAPE '" & objShape.Name & "': """ & Trim(Left(visText, 50)) & """  Latin=" & visLatin & "/" & visNW & "(" & Format(visPctLatin * 100, "0") & "%)  -> " & IIf(pptLang = "ZH", "EN", "ZH") & vbCrLf
                                End If
                            End If
                        End If
                    End If
                End If
SkipVisShape:
            Next objShape
            If visSlideGaps > 0 Then
                visGapSlides = visGapSlides + 1
                If gapShapeSlides <> "" Then gapShapeSlides = gapShapeSlides & ", "
                gapShapeSlides = gapShapeSlides & slideNum
                slideGapLog = slideGapLog & "      (Visible text: " & visSlideGaps & " shape(s))" & vbCrLf & visGapLog
            End If
        End If

        ' Notes whole-text check (all slides)
        Dim notesForTrans As String
        notesForTrans = GetNotesText(sld)
        If notesForTrans <> "" Then
            Dim noteText As String
            noteText = Trim(notesForTrans)
            ' Skip if the entire note is a technical fragment
            If Not techFragRegex.Test(noteText) Then
                Dim noteLang As String
                noteLang = DetectLanguage(noteText)
                If noteLang <> "" And noteLang <> pptLang Then
                    notesGapCount = notesGapCount + 1
                    hasGap = True
                    If gapNoteSlides <> "" Then gapNoteSlides = gapNoteSlides & ", "
                    gapNoteSlides = gapNoteSlides & slideNum
                    If DEBUG_MODE Then
                        slideGapLog = slideGapLog & "      NOTES: '" & Trim(Left(noteText, 120)) & "' [" & noteLang & "]" & vbCrLf
                    End If
                End If
            End If
        End If

        If hasGap Then
            transGapLog = transGapLog & "  SLIDE " & slideNum & " (Title: '" & slides(slideNum).OrigTitle & "')" & vbCrLf & slideGapLog
        End If
    Next slideNum

    transSummary = "TRANSLATION GAP CHECK" & vbCrLf & _
                   "  Language: " & pptLang & " | Shapes checked: " & visShapesChecked & " | Gaps: " & shapeGapCount & " shape(s), " & notesGapCount & " note(s)" & vbCrLf
    transGapNoteSlides = gapNoteSlides
    transGapShapeSlides = gapShapeSlides
    If shapeGapCount > 0 Or notesGapCount > 0 Then
        If DEBUG_MODE Then
            transSummary = transSummary & transGapLog & vbCrLf
        Else
            ' Normal mode: show which slides have note gaps (no sentence detail)
            If notesGapCount > 0 Then
                transSummary = transSummary & "  Note gaps on slides: " & transGapNoteSlides & vbCrLf
            End If
            If shapeGapCount > 0 Then
                transSummary = transSummary & "  Shape gaps on slides: " & transGapShapeSlides & vbCrLf
            End If
        End If
    Else
        transSummary = transSummary & "  No gaps detected." & vbCrLf
    End If
End Sub

Private Sub BuildSkippedSummary(ByRef slides() As SlideData, _
                                ByVal totalSlides As Long, _
                                ByRef skippedSummary As String)
    Dim reasonList() As String: ReDim reasonList(1 To totalSlides)
    Dim uniqueReasonCount As Long: uniqueReasonCount = 0
    Dim s As Long, u As Long
    Dim found As Boolean

    For s = 1 To totalSlides
        If slides(s).Skip And slides(s).SkipReason <> "" Then
            found = False
            For u = 1 To uniqueReasonCount
                If reasonList(u) = slides(s).SkipReason Then
                    found = True
                    Exit For
                End If
            Next u
            If Not found Then
                uniqueReasonCount = uniqueReasonCount + 1
                reasonList(uniqueReasonCount) = slides(s).SkipReason
            End If
        End If
    Next s

    If uniqueReasonCount > 0 Then
        skippedSummary = "SKIPPED SLIDES SUMMARY" & vbCrLf
        For u = 1 To uniqueReasonCount
            Dim nums As String: nums = ""
            For s = 1 To totalSlides
                If slides(s).Skip And slides(s).SkipReason = reasonList(u) Then
                    If nums <> "" Then nums = nums & ", "
                    nums = nums & s
                End If
            Next s
            skippedSummary = skippedSummary & "  [" & reasonList(u) & "] Slides: " & nums & vbCrLf
        Next u
        skippedSummary = skippedSummary & vbCrLf
    End If
End Sub

Private Sub BuildDebugMatrix(ByRef slides() As SlideData, _
                             ByVal totalSlides As Long, _
                             ByRef matrixLog As String)
    ' COMPACT MODE: only show slides where something changed
    ' (title differs from original, or slide was skipped)
    ' To see all slides, set fullMatrix = True below.
    Dim fullMatrix As Boolean: fullMatrix = False
    Dim slideNum As Long
    matrixLog = "DEBUG INSPECTION MATRIX" & vbCrLf

    Dim shownCount As Long: shownCount = 0
    For slideNum = 1 To totalSlides
        Dim showThis As Boolean: showThis = fullMatrix
        If Not fullMatrix Then
            If slides(slideNum).Skip Then
                showThis = True
            ElseIf slides(slideNum).OrigTitle <> slides(slideNum).FinalTitle Then
                showThis = True
            End If
        End If
        If showThis Then
            shownCount = shownCount + 1
            matrixLog = matrixLog & "SLIDE " & slideNum & ":" & vbCrLf & _
                        "  - Checked Shapes Inventory : " & slides(slideNum).CheckedShapes & vbCrLf & _
                        "  - Evaluated Text Blocks    : " & slides(slideNum).DiscoveredTexts & vbCrLf

            If slides(slideNum).Skip Then
                matrixLog = matrixLog & "  - SELECTION RESULT         : [SKIPPED - " & slides(slideNum).SkipReason & "]" & vbCrLf
            Else
                matrixLog = matrixLog & "  - TIE-BREAKER WINNER SHAPE : [" & slides(slideNum).ShapeName & "]" & vbCrLf & _
                            "  - FINAL SELECTION RESULT   : " & vbCrLf & _
                            "      * Scanned Ingestion -> '" & slides(slideNum).OrigTitle & "'" & vbCrLf & _
                            "      * Committed Final   -> '" & slides(slideNum).FinalTitle & "'" & vbCrLf
            End If
            matrixLog = matrixLog & "--------------------------------------------------" & vbCrLf
        End If
    Next slideNum

    If shownCount < totalSlides Then
        matrixLog = matrixLog & "Showing " & shownCount & " of " & totalSlides & " slides (changed/skipped only)." & vbCrLf
        matrixLog = matrixLog & "Set fullMatrix = True in BuildDebugMatrix to see all slides." & vbCrLf & vbCrLf
    End If
End Sub

' ===================================================================
' PHASE: ProcessFixableChanges
' Called after AnalyseDuplicates. Builds consecutive-slide groups
' from array diffs, prompts the user one group at a time, and
' applies accepted changes to the actual presentation shapes.
'
' Yes  → update shape text, keep array changes (committed).
' No   → revert arrays to backup for that group (rolled back).
' Cancel → revert all remaining groups and exit.
'
' Returns True if ANY slide arrays were reverted to backup
' (so the caller knows to re-run AnalyseDuplicates for accurate counts).
' ===================================================================

Private Function ProcessFixableChanges(ByRef origPres As Presentation, _
                                       ByRef slides() As SlideData, _
                                       ByRef backupSlides() As SlideData, _
                                       ByVal totalSlides As Long) As Boolean
    Dim slideNum As Long
    Dim groupStart() As Long, groupEnd() As Long
    Dim groupCount As Long
    Dim i As Long, g As Long, s As Long

    ' Build groups of consecutive slides where FinalTitle differs from OrigTitle
    ReDim groupStart(1 To totalSlides)
    ReDim groupEnd(1 To totalSlides)
    groupCount = 0

    slideNum = 1
    Do While slideNum <= totalSlides
        If Not slides(slideNum).Skip And slides(slideNum).FinalTitle <> slides(slideNum).OrigTitle Then
            groupCount = groupCount + 1
            groupStart(groupCount) = slideNum
            Do While slideNum <= totalSlides And Not slides(slideNum).Skip And slides(slideNum).FinalTitle <> slides(slideNum).OrigTitle
                slideNum = slideNum + 1
            Loop
            groupEnd(groupCount) = slideNum - 1
        Else
            slideNum = slideNum + 1
        End If
    Loop

    If groupCount = 0 Then
        ProcessFixableChanges = False
        Exit Function
    End If

    ' Ask user whether they want to review
    Dim msg As String
    msg = "Fixable changes detected (" & groupCount & " group(s)):" & vbCrLf & vbCrLf
    msg = msg & "Would you like to review and apply them one group at a time?"

    If MsgBox(msg, vbYesNo + vbQuestion, "Fixable Changes Detected") = vbNo Then
        ProcessFixableChanges = False
        Exit Function
    End If

    ' Process each group
    Dim appliedCount As Long: appliedCount = 0
    Dim skippedCount As Long: skippedCount = 0
    Dim anyReverted As Boolean: anyReverted = False
    Dim sld As Slide
    Dim objShape As Shape

    For g = 1 To groupCount
        Dim startS As Long: startS = groupStart(g)
        Dim endS As Long: endS = groupEnd(g)
        Dim groupSize As Long: groupSize = endS - startS + 1

        ' Build prompt for this group — cap at ~900 chars to avoid MsgBox overflow
        Dim prompt As String
        If groupSize = 1 Then
            prompt = "Slide " & startS & ":" & vbCrLf & vbCrLf
        Else
            prompt = "Slides " & startS & "-" & endS & " (" & groupSize & " slides):" & vbCrLf & vbCrLf
        End If

        Dim shownCount As Long: shownCount = 0
        For s = startS To endS
            If Len(prompt) > 900 Then
                prompt = prompt & "  ... and " & (groupSize - shownCount) & " more change(s)" & vbCrLf
                Exit For
            End If
            prompt = prompt & "  Slide " & s & ": '" & slides(s).OrigTitle & "'" & vbCrLf
            prompt = prompt & "         -> '" & slides(s).FinalTitle & "'" & vbCrLf
            shownCount = shownCount + 1
        Next s

        prompt = prompt & vbCrLf & "Apply these " & groupSize & " change(s)?"

        Dim response As VbMsgBoxResult
        response = MsgBox(prompt, vbYesNoCancel + vbQuestion, _
                          "Fixable Change Group " & g & " of " & groupCount)

        If response = vbCancel Then
            ' Revert all remaining groups (including current)
            For i = g To groupCount
                For s = groupStart(i) To groupEnd(i)
                    slides(s) = backupSlides(s)
                Next s
                skippedCount = skippedCount + (groupEnd(i) - groupStart(i) + 1)
            Next i
            anyReverted = True
            Exit For

        ElseIf response = vbYes Then
            ' Apply every slide in this group
            For s = startS To endS
                Set sld = origPres.Slides(s)
                If slides(s).ShapeName <> "" Then
                    Set objShape = Nothing
                    On Error Resume Next
                    Set objShape = sld.Shapes(slides(s).ShapeName)
                    On Error GoTo 0

                    If Not objShape Is Nothing Then
                        If objShape.HasTextFrame And objShape.TextFrame.HasText Then
                            Dim textToAssign As String
                            textToAssign = slides(s).FinalTitle

                            On Error Resume Next
                            If slides(s).MatchLength <= objShape.TextFrame.TextRange.Length Then
                                If slides(s).MatchLength > 0 Then
                                    objShape.TextFrame.TextRange.Characters(1, slides(s).MatchLength).Text = textToAssign
                                Else
                                    objShape.TextFrame.TextRange.Text = textToAssign
                                End If
                            Else
                                objShape.TextFrame.TextRange.Text = textToAssign
                            End If

                            If Err.Number = 0 Then
                                appliedCount = appliedCount + 1
                            Else
                                Err.Clear
                            End If
                            On Error GoTo 0
                        End If
                    End If
                End If
            Next s

        Else ' vbNo — reject the whole group, revert arrays to backup
            For s = startS To endS
                slides(s) = backupSlides(s)
            Next s
            anyReverted = True
            skippedCount = skippedCount + groupSize
        End If
    Next g

    ' Save if any changes were applied
    If appliedCount > 0 Then
        On Error Resume Next
        origPres.Save
        If Err.Number <> 0 Then
            MsgBox "Applied " & appliedCount & " change(s), but failed to save:" & vbCrLf & _
                   Err.Description & vbCrLf & vbCrLf & _
                   "Save the presentation manually.", vbExclamation, "Save Failed"
            Err.Clear
        Else
            MsgBox "Applied " & appliedCount & " change(s)." & vbCrLf & _
                   "Skipped: " & skippedCount & "." & vbCrLf & vbCrLf & _
                   "Presentation has been saved.", vbInformation, "Changes Applied"
        End If
        On Error GoTo 0
    Else
        MsgBox "No changes were applied.", vbInformation, "No Changes"
    End If

    ProcessFixableChanges = anyReverted
End Function

' ===================================================================
' FLATTENING ACTION: FlattenPresentation
' Extracts titles, exports PNGs, creates flattened copy.
' Assumes the user has fixed any issues found by RunAllChecks.
' ===================================================================

Private Sub FlattenPresentation()
    Dim sld As Slide, shp As Shape
    Dim pptApp As PowerPoint.Application
    Dim origPres As Presentation
    Dim newPres As Presentation
    Dim origPath As String, origName As String
    Dim tempFolder As String, newPresPath As String
    Dim imgPath As String
    Dim FSO As Object
    Dim slideNum As Long, i As Long
    Dim result As AnalysisResult

    On Error GoTo ErrorHandler

    Set pptApp = PowerPoint.Application
    Set origPres = pptApp.ActivePresentation

    If origPres.Path = "" Then
        MsgBox "Please save your original presentation before running this script.", vbCritical, "Save Required"
        Exit Sub
    End If
    If origPres.Slides.Count = 0 Then
        MsgBox "Presentation has no slides.", vbExclamation, "Nothing to Flatten"
        Exit Sub
    End If

    ' Extract titles (captureDebug = False — no debug arrays needed)
    result = PopulateTitleArrays(origPres, captureDebug:=False)

    ' Initialize paths
    origPath = origPres.Path & "\"
    Dim dotPos As Long: dotPos = InStrRev(origPres.Name, ".")
    If dotPos > 1 Then
        origName = Left(origPres.Name, dotPos - 1)
    Else
        origName = origPres.Name
    End If
    tempFolder = origPath & "PPT_Temp_Images_Internal\"
    newPresPath = origPath & origName & "_Flattened.pptx"

    Set FSO = CreateObject("Scripting.FileSystemObject")

    ' ==================================================
    ' STEP 2: HIGH RESOLUTION IMAGE EXPORT
    ' ==================================================

    If Not FSO.FolderExists(tempFolder) Then
        FSO.CreateFolder tempFolder
    End If

    For Each sld In origPres.Slides
        imgPath = tempFolder & "Slide_" & sld.SlideIndex & ".png"
        sld.Export imgPath, "PNG", result.SlideWidth * 2, result.SlideHeight * 2
    Next sld

    ' Warn if flattened copy already exists (will be overwritten)
    If FSO.FileExists(newPresPath) Then
        If MsgBox("A previous flattened copy already exists:" & vbCrLf & vbCrLf & _
                  newPresPath & vbCrLf & vbCrLf & _
                  "It will be overwritten. Continue?", _
                  vbYesNo + vbExclamation, "Overwrite Existing File") = vbNo Then
            ' Clean up exported images and exit
            On Error Resume Next
            FSO.DeleteFolder Left(tempFolder, Len(tempFolder) - 1), True
            On Error GoTo ErrorHandler
            Exit Sub
        End If
    End If

    origPres.SaveCopyAs newPresPath
    Set newPres = pptApp.Presentations.Open(newPresPath, WithWindow:=msoTrue)

    ' ==================================================
    ' STEP 3: FLATTEN LOOP
    ' ==================================================

    For Each sld In newPres.Slides
        slideNum = sld.SlideIndex

        If result.Slides(slideNum).Skip Then GoTo SkipFlattenSlide

        ' Dynamic text modifications pass on parent tracking shapes
        If result.Slides(slideNum).FinalTitle <> "" Then
            Dim normalizedOriginal As String, normalizedFinal As String
            normalizedOriginal = Replace(result.Slides(slideNum).OrigTitle, " ", "")
            normalizedFinal = Replace(result.Slides(slideNum).FinalTitle, " ", "")

            If normalizedOriginal <> normalizedFinal Then
                If result.Slides(slideNum).ShapeName <> "" Then
                    Dim objShape As Shape
                    Set objShape = Nothing
                    On Error Resume Next
                    Set objShape = sld.Shapes(result.Slides(slideNum).ShapeName)
                    If Not objShape Is Nothing Then
                        If objShape.HasTextFrame Then
                            Dim leftAnchorPos As Single
                            leftAnchorPos = objShape.Left

                            objShape.TextFrame.TextRange.ParagraphFormat.Alignment = ppAlignLeft
                            objShape.TextFrame.MarginLeft = 0
                            objShape.TextFrame.MarginRight = 0
                            objShape.TextFrame.WordWrap = msoFalse
                            objShape.TextFrame.AutoSize = msoAutoSizeShapeToFitText

                            Dim textToAssign As String: textToAssign = result.Slides(slideNum).FinalTitle
                            If result.Slides(slideNum).MatchLength <= objShape.TextFrame.TextRange.Length Then
                                If result.Slides(slideNum).MatchLength > 0 Then
                                    objShape.TextFrame.TextRange.Characters(1, result.Slides(slideNum).MatchLength).Text = textToAssign
                                Else
                                    objShape.TextFrame.TextRange.Text = textToAssign
                                End If
                            Else
                                objShape.TextFrame.TextRange.Text = textToAssign
                            End If
                            objShape.Left = leftAnchorPos
                        End If
                    End If
                    On Error GoTo ErrorHandler
                End If
            End If
        End If

        ' Clean native layout shape vector trees — preserve media shapes (audio/video)
        For i = sld.Shapes.Count To 1 Step -1
            If sld.Shapes(i).Type <> msoMedia Then
                sld.Shapes(i).Delete
            End If
        Next i

        imgPath = tempFolder & "Slide_" & sld.SlideIndex & ".png"

        If FSO.FileExists(imgPath) Then
            Set shp = sld.Shapes.AddPicture(FileName:=imgPath, _
                                            LinkToFile:=msoFalse, _
                                            SaveWithDocument:=msoTrue, _
                                            Left:=0, _
                                            Top:=0, _
                                            Width:=result.SlideWidth, _
                                            Height:=result.SlideHeight)

            With shp
                .LockAspectRatio = msoFalse
                On Error Resume Next
                .PictureFormat.CropLeft = 0
                .PictureFormat.CropTop = 0
                .PictureFormat.CropRight = 0
                .PictureFormat.CropBottom = 0
                On Error GoTo ErrorHandler
                .Left = 0
                .Top = 0
                .Width = result.SlideWidth
                .Height = result.SlideHeight
            End With

            ' Structural Native Title Placeholder Injection
            ' Wrapped in On Error Resume Next so a single bad slide
            ' does not roll back the entire flatten.
            If result.Slides(slideNum).FinalTitle <> "" And result.Slides(slideNum).Skip = False Then
                On Error Resume Next
                sld.Layout = ppLayoutTitleOnly
                Dim oTargetTitlePlaceholder As Shape
                Set oTargetTitlePlaceholder = sld.Shapes.Title

                If Err.Number = 0 And Not oTargetTitlePlaceholder Is Nothing Then
                    With oTargetTitlePlaceholder
                        .Name = result.Slides(slideNum).FinalTitle
                        With .TextFrame.TextRange
                            .Text = result.Slides(slideNum).FinalTitle
                            .Font.Size = 1
                            .Font.Color.RGB = RGB(255, 255, 255)
                        End With
                        .Fill.Transparency = 1#
                        .Line.Transparency = 1#
                        .ZOrder msoSendToBack
                    End With
                End If
                Err.Clear
                On Error GoTo ErrorHandler
            End If
        End If

        ' Commit explicit programmatic slide name references with collision guardrail
        If result.Slides(slideNum).FinalTitle <> "" And result.Slides(slideNum).Skip = False Then
            Dim uniqueSlideName As String, testSlide As Slide
            Dim nameCollisionCounter As Long

            uniqueSlideName = result.Slides(slideNum).FinalTitle
            nameCollisionCounter = 1

            Do
                Set testSlide = Nothing
                On Error Resume Next
                Set testSlide = newPres.Slides(uniqueSlideName)
                On Error GoTo ErrorHandler

                If testSlide Is Nothing Then
                    Exit Do
                ElseIf testSlide.SlideIndex = sld.SlideIndex Then
                    Exit Do
                ElseIf nameCollisionCounter >= 999 Then
                    ' Safety limit — avoid infinite loop
                    Exit Do
                Else
                    nameCollisionCounter = nameCollisionCounter + 1
                    uniqueSlideName = result.Slides(slideNum).FinalTitle & "_" & nameCollisionCounter
                End If
            Loop

            sld.Name = uniqueSlideName
        End If
SkipFlattenSlide:
    Next sld

    newPres.Save

    ' View restore — target flattened copy
    On Error Resume Next
    newPres.Windows(1).Activate
    ActiveWindow.View.GotoSlide 1
    ActiveWindow.View.ZoomToFit = msoTrue
    On Error GoTo ErrorHandler

    Set newPres = Nothing   ' success — protect from error handler rollback close

    ' Close the original (suppress save prompt)
    origPres.Saved = msoTrue
    origPres.Close
    Set origPres = Nothing

    ' Clean up temp images
    If FSO.FolderExists(tempFolder) Then
        On Error Resume Next
        FSO.DeleteFolder Left(tempFolder, Len(tempFolder) - 1), True
        On Error GoTo ErrorHandler
    End If

    Dim skippedSlidesLog As String: skippedSlidesLog = ""
    Dim skippedTotal As Long: skippedTotal = 0
    For slideNum = 1 To result.TotalSlides
        If result.Slides(slideNum).Skip Then
            skippedTotal = skippedTotal + 1
            skippedSlidesLog = skippedSlidesLog & "  Slide " & slideNum & " [" & result.Slides(slideNum).SkipReason & "]" & vbCrLf
        End If
    Next slideNum

    Dim briefLog As String
    briefLog = "==================================================" & vbCrLf & _
               "FLATTEN REPORT" & vbCrLf & _
               "==================================================" & vbCrLf & _
               "TIMESTAMP: " & Now & vbCrLf & _
               "TOTAL SLIDES: " & result.TotalSlides & vbCrLf & _
               "FLATTENED: " & (result.TotalSlides - skippedTotal) & vbCrLf & _
               "SKIPPED: " & skippedTotal & vbCrLf & _
               "==================================================" & vbCrLf & vbCrLf
    If skippedTotal > 0 Then
        briefLog = briefLog & "Skipped slides:" & vbCrLf & skippedSlidesLog & vbCrLf
    End If

    WriteLog briefLog, origPath

    MsgBox "Flattening complete. " & (result.TotalSlides - skippedTotal) & " slides flattened.", vbInformation, "Process Complete"
    Exit Sub

ErrorHandler:
    On Error Resume Next
    If Not newPres Is Nothing Then
        newPres.Saved = msoTrue
        newPres.Close
    End If
    If Not FSO Is Nothing Then
        If FSO.FolderExists(tempFolder) Then
            FSO.DeleteFolder Left(tempFolder, Len(tempFolder) - 1), True
        End If
    End If
    MsgBox "An execution interruption occurred. Environment rolled back safely.", vbCritical, "System Failure"
End Sub
