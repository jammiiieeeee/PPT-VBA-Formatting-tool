' ==================================================
' PPT Translation Gap Diagnostic
' Checks notes AND visible text for missed translations
' against the presentation-level language.
' Paste into VBA editor (Alt+F11 -> Insert -> Module)
' then Alt+F8 -> run PPTTools_TestTransGaps
' ==================================================

Option Explicit

Private mSldHeight As Single

Public Sub PPTTools_TestTransGaps()
    Dim pptApp As PowerPoint.Application
    Dim pres As Presentation
    Dim sld As Slide
    Dim slideNum As Long
    Dim log As String, logGaps As String, logOther As String
    Dim totalScanned As Long, totalWithNotes As Long, totalGapSlides As Long
    Dim totalSkippedNoNotes As Long
    Dim totalSentences As Long, totalGaps As Long
    Dim totalVisShapesChecked As Long, totalVisGapShapes As Long, totalVisGapSlides As Long

    Set pptApp = PowerPoint.Application
    Set pres = pptApp.ActivePresentation
    If pres.Path = "" Then
        MsgBox "Please save the presentation first.", vbCritical, "Save Required"
        Exit Sub
    End If

    mSldHeight = pres.PageSetup.SlideHeight

    ' Pre-scan: build combined title text (text portion only, strip numbering index)
    Dim combinedTitles As String
    combinedTitles = ""
    Dim stripRegex As Object
    Set stripRegex = CreateObject("VBScript.RegExp")
    stripRegex.Pattern = "^\d+(?:[\.\(/\\\-_\??]\d+)*\s*(.*)"
    stripRegex.IgnoreCase = True
    For slideNum = 1 To pres.Slides.Count
        Set sld = pres.Slides(slideNum)
        Dim t As String
        t = GetTitleText(sld)
        If t <> "" Then
            Dim textOnly As String
            If stripRegex.Test(t) Then
                textOnly = Trim(stripRegex.Execute(t)(0).SubMatches(0))
            Else
                textOnly = t
            End If
            If textOnly <> "" Then combinedTitles = combinedTitles & " " & textOnly
        End If
    Next slideNum

    Dim pptLang As String
    pptLang = DetectLanguage(combinedTitles)
    ' Fallback: if combined text is MIXED, try per-title majority vote
    If pptLang = "" Or pptLang = "MIXED" Then
        Dim zhCount As Long, enCount As Long
        zhCount = 0: enCount = 0
        For slideNum = 1 To pres.Slides.Count
            Set sld = pres.Slides(slideNum)
            t = GetTitleText(sld)
            If t <> "" Then
                If stripRegex.Test(t) Then
                    textOnly = Trim(stripRegex.Execute(t)(0).SubMatches(0))
                Else
                    textOnly = t
                End If
                If textOnly <> "" Then
                    Dim lang As String
                    lang = DetectLanguage(textOnly)
                    If lang = "ZH" Then zhCount = zhCount + 1
                    If lang = "EN" Then enCount = enCount + 1
                End If
            End If
        Next slideNum
        If zhCount > enCount Then
            pptLang = "ZH"
        ElseIf enCount > zhCount Then
            pptLang = "EN"
        Else
            MsgBox "Cannot determine presentation language." & vbCrLf & vbCrLf & _
                   "Combined text: """ & Left(combinedTitles, 200) & """" & vbCrLf & _
                   "ZH titles: " & zhCount & ", EN titles: " & enCount, vbCritical, "Language Detection Failed"
            Exit Sub
        End If
    End If

    ' Main scan
    log = "==================================================" & vbCrLf
    log = log & "NOTES TRANSLATION GAP DIAGNOSTIC" & vbCrLf
    log = log & "==================================================" & vbCrLf
    log = log & "Presentation: " & pres.Name & vbCrLf
    log = log & "Presentation language: " & pptLang & vbCrLf
    log = log & "TIMESTAMP: " & Now & vbCrLf
    log = log & "==================================================" & vbCrLf & vbCrLf

    ' Summary placeholder — will be filled after loop
    Dim summaryBlock As String

    For slideNum = 1 To pres.Slides.Count
        Set sld = pres.Slides(slideNum)

        Dim titleText As String, notesText As String
        titleText = GetTitleText(sld)
        notesText = GetNotesText_Repro(sld)
        totalScanned = totalScanned + 1

        Dim slideBlock As String
        slideBlock = "--- Slide " & slideNum & " ---" & vbCrLf
        If titleText <> "" Then
            slideBlock = slideBlock & "  Title: """ & Trim(Left(titleText, 80)) & """" & vbCrLf
        Else
            slideBlock = slideBlock & "  Title: (none found)" & vbCrLf
        End If
        slideBlock = slideBlock & "  Language: " & pptLang & " (ppt-level)" & vbCrLf

        ' --- Visible text shape scan (full slide, all visible text shapes) ---
        Dim visGapLog As String, visGapCount As Long, visShp As Object
        visGapLog = ""
        visGapCount = 0
        For Each visShp In sld.Shapes
            If visShp.Visible And visShp.HasTextFrame And visShp.TextFrame.HasText Then
                Dim visText As String
                visText = Trim(visShp.TextFrame.TextRange.Text)
                If Len(visText) > 0 Then
                    Dim visNW As Long, visNonLatin As Long, visJ As Long, visC As String, visCp As Long
                    visNW = 0: visNonLatin = 0
                    For visJ = 1 To Len(visText)
                        visC = Mid(visText, visJ, 1)
                        If visC <> " " And visC <> vbCr And visC <> vbLf And visC <> vbTab Then
                            visNW = visNW + 1
                            visCp = AscW(visC)
                            If Not ((visCp >= 65 And visCp <= 90) Or (visCp >= 97 And visCp <= 122)) Then
                                visNonLatin = visNonLatin + 1
                            End If
                        End If
                    Next visJ
                    If visNW >= 5 Then
                        totalVisShapesChecked = totalVisShapesChecked + 1
                        ' Skip shape that matches the slide title (case-insensitive)
                        If titleText <> "" And StrComp(visText, Trim(titleText), vbTextCompare) = 0 Then GoTo SkipTitleShape
                        Dim visLatin As Long
                        visLatin = visNW - visNonLatin
                        Dim visPctLatin As Double, visPctNonLatin As Double
                        visPctLatin = visLatin / visNW
                        visPctNonLatin = visNonLatin / visNW

                        Dim visIsGap As Boolean
                        visIsGap = False
                        If pptLang = "ZH" And visPctLatin >= 0.9 Then visIsGap = True
                        If pptLang = "EN" And visPctNonLatin >= 0.9 Then visIsGap = True

                        If visIsGap Then
                            visGapCount = visGapCount + 1
                            totalVisGapShapes = totalVisGapShapes + 1
                            visGapLog = visGapLog & "    [" & visShp.Name & "] """ & Trim(Left(visText, 50)) & """  Latin=" & visLatin & "/" & visNW & "(" & Format(visPctLatin * 100, "0") & "%) CJK=" & visNonLatin & "/" & visNW & "(" & Format(visPctNonLatin * 100, "0") & "%)  -> " & IIf(pptLang = "ZH", "EN", "ZH") & " *** MISSED TRANSLATION" & vbCrLf
                        End If
SkipTitleShape:
                    End If
                End If
            End If
        Next visShp
        If visGapCount > 0 Then totalVisGapSlides = totalVisGapSlides + 1

        ' --- Notes analysis ---
        Dim noteGapCount As Long, noteSentLog As String
        noteGapCount = 0
        noteSentLog = ""

        If notesText <> "" Then
            totalWithNotes = totalWithNotes + 1
            slideBlock = slideBlock & "  Notes: """ & Trim(Left(notesText, 100)) & """" & vbCrLf

            ' Split notes into sentences
            Dim temp As String
            temp = Replace(notesText, "!", ".")
            temp = Replace(temp, "?", ".")
            temp = Replace(temp, "。", ".")
            temp = Replace(temp, "！", ".")
            temp = Replace(temp, "？", ".")
            Dim sentences() As String
            sentences = Split(temp, ".")

            Dim si As Long
            For si = 0 To UBound(sentences)
                Dim s As String
                s = Trim(sentences(si))
                If Len(s) > 0 Then
                    totalSentences = totalSentences + 1

                    Dim sLang As String
                    sLang = DetectLanguage(s)

                    ' Character breakdown
                    Dim cjk As Long, latin As Long, ci As Long, c As String, code As Long
                    cjk = 0: latin = 0
                    For ci = 1 To Len(s)
                        c = Mid(s, ci, 1)
                        If c <> " " And c <> vbCr And c <> vbLf And c <> vbTab Then
                            code = AscW(c)
                            If (code >= 65 And code <= 90) Or (code >= 97 And code <= 122) Then
                                latin = latin + 1
                            Else
                                cjk = cjk + 1
                            End If
                        End If
                    Next ci
                    Dim totalChars As Long
                    totalChars = cjk + latin
                    Dim pctCJK As String, pctLatin As String
                    If totalChars > 0 Then
                        pctCJK = Format(cjk / totalChars * 100, "0")
                        pctLatin = Format(latin / totalChars * 100, "0")
                    Else
                        pctCJK = "0"
                        pctLatin = "0"
                    End If

                    If sLang <> "" And sLang <> pptLang Then
                        noteGapCount = noteGapCount + 1
                        totalGaps = totalGaps + 1
                        noteSentLog = noteSentLog & "    [" & (si + 1) & "/" & (UBound(sentences) + 1) & "] """ & Trim(Left(s, 60)) & """  CJK=" & cjk & "(" & pctCJK & "%) Latin=" & latin & "(" & pctLatin & "%)  -> " & sLang & " *** GAP" & vbCrLf
                    End If
                End If
            Next si

            If noteGapCount > 0 Then
                totalGapSlides = totalGapSlides + 1
                slideBlock = slideBlock & "  Status: " & noteGapCount & " gap(s)" & vbCrLf
                slideBlock = slideBlock & "  Sentences:" & vbCrLf
                slideBlock = slideBlock & noteSentLog
            Else
                slideBlock = slideBlock & "  Status: OK (0 gaps)" & vbCrLf
            End If
        Else
            totalSkippedNoNotes = totalSkippedNoNotes + 1
            slideBlock = slideBlock & "  Status: SKIPPED (no notes)" & vbCrLf
        End If

        ' Add visible text section
        If visGapCount > 0 Then
            slideBlock = slideBlock & "  Visible text: " & visGapCount & " gap(s)" & vbCrLf
            slideBlock = slideBlock & visGapLog
        End If

        ' Route: gaps (notes OR visible) → logGaps, else → logOther
        If noteGapCount > 0 Or visGapCount > 0 Then
            logGaps = logGaps & slideBlock & vbCrLf
        Else
            logOther = logOther & slideBlock & vbCrLf
        End If
    Next slideNum

    ' Assemble: header → summary → gaps → others
    summaryBlock = "==================================================" & vbCrLf
    summaryBlock = summaryBlock & "SUMMARY" & vbCrLf
    summaryBlock = summaryBlock & "==================================================" & vbCrLf
    summaryBlock = summaryBlock & "Total slides:                " & totalScanned & vbCrLf
    summaryBlock = summaryBlock & "Presentation language:       " & pptLang & vbCrLf
    summaryBlock = summaryBlock & "  Notes:" & vbCrLf
    summaryBlock = summaryBlock & "    - No notes:              " & totalSkippedNoNotes & vbCrLf
    summaryBlock = summaryBlock & "    - Analyzed:              " & totalWithNotes & vbCrLf
    summaryBlock = summaryBlock & "      - With gaps:           " & totalGapSlides & vbCrLf
    summaryBlock = summaryBlock & "      - Clean:               " & (totalWithNotes - totalGapSlides) & vbCrLf
    summaryBlock = summaryBlock & "    Total sentences:          " & totalSentences & vbCrLf
    summaryBlock = summaryBlock & "    Total note gaps:          " & totalGaps & vbCrLf
    summaryBlock = summaryBlock & "  Visible text:" & vbCrLf
    summaryBlock = summaryBlock & "    - Shapes checked:        " & totalVisShapesChecked & vbCrLf
    summaryBlock = summaryBlock & "    - Slides with gaps:      " & totalVisGapSlides & vbCrLf
    summaryBlock = summaryBlock & "    - Missed translations:   " & totalVisGapShapes & vbCrLf
    summaryBlock = summaryBlock & "==================================================" & vbCrLf & vbCrLf

    log = log & summaryBlock

    If logGaps <> "" Then
        log = log & "===================== GAPS FOUND =====================" & vbCrLf & vbCrLf
        log = log & logGaps
        log = log & "==================================================" & vbCrLf & vbCrLf
    End If
    log = log & "===================== OTHER SLIDES ====================" & vbCrLf & vbCrLf
    log = log & logOther

    Dim wsh As Object, fileNum As Integer, logPath As String
    Set wsh = CreateObject("WScript.Shell")
    logPath = wsh.SpecialFolders("Desktop") & "\PPT_Trans_Gap_Test_" & Format(Now, "yyyymmdd_hhmmss") & ".txt"
    Set wsh = Nothing
    fileNum = FreeFile
    Open logPath For Output As #fileNum
    Print #fileNum, log
    Close #fileNum

    Dim gapMsg As String
    gapMsg = "Slides w/ note gaps: " & totalGapSlides & " (" & totalGaps & " gaps)" & vbCrLf
    gapMsg = gapMsg & "Slides w/ vis text gaps: " & totalVisGapSlides & " (" & totalVisGapShapes & " shapes)"
    MsgBox "Diagnostic complete." & vbCrLf & vbCrLf & gapMsg, vbInformation, "Trans Gap Test Done"
End Sub

' ==================================================
' Get title text from slide top half via regex + depth
' ==================================================
Private Function GetTitleText(ByRef sld As Slide) As String
    Dim regex As Object, matches As Object, shp As Object
    Dim fullText As String, textBlocks() As String, blockIdx As Long
    Dim currentBlockText As String, currentIndex As String, currentText As String
    Dim remainderStr As String, currentDots As Long
    Dim deepestIndex As String, deepestText As String, maxDots As Long

    Set regex = CreateObject("VBScript.RegExp")
    With regex
        .Pattern = "^(\d+(?:[\.\(/\\\-_\??]\d+)*)\s*(.*)"
        .IgnoreCase = True
        .Global = True
    End With

    deepestIndex = "": deepestText = "": maxDots = -1

    For Each shp In sld.Shapes
        If shp.Visible And shp.HasTextFrame And shp.TextFrame.HasText Then
            fullText = shp.TextFrame.TextRange.Text
            fullText = Replace(fullText, vbCrLf, vbCr)
            fullText = Replace(fullText, vbLf, vbCr)
            textBlocks = Split(fullText, vbCr)

            For blockIdx = 0 To UBound(textBlocks)
                currentBlockText = Trim(textBlocks(blockIdx))
                If Len(currentBlockText) > 0 And shp.Top < mSldHeight * 0.5 Then
                    If regex.Test(currentBlockText) Then
                        Set matches = regex.Execute(currentBlockText)
                        currentIndex = matches(0).SubMatches(0)
                        currentText = matches(0).SubMatches(1)

                        remainderStr = currentIndex
                        remainderStr = Replace(remainderStr, ".", "")
                        remainderStr = Replace(remainderStr, "(", "")
                        remainderStr = Replace(remainderStr, "-", "")
                        remainderStr = Replace(remainderStr, "_", "")
                        remainderStr = Replace(remainderStr, "/", "")
                        remainderStr = Replace(remainderStr, "\", "")
                        remainderStr = Replace(remainderStr, "?", "")
                        currentDots = Len(currentIndex) - Len(remainderStr)

                        If currentDots > maxDots Then
                            maxDots = currentDots
                            deepestIndex = currentIndex
                            deepestText = currentText
                        End If
                    End If
                End If
            Next blockIdx
        End If
    Next shp

    If deepestIndex <> "" Then
        If Len(Trim(deepestText)) > 0 Then
            GetTitleText = deepestIndex & " " & Trim(deepestText)
        Else
            GetTitleText = deepestIndex
        End If
    Else
        ' Fallback: first text from any visible shape in top half
        For Each shp In sld.Shapes
            If shp.Visible And shp.HasTextFrame And shp.TextFrame.HasText Then
                If shp.Top < mSldHeight * 0.5 Then
                    Dim fallback As String
                    fallback = Trim(shp.TextFrame.TextRange.Text)
                    If Len(fallback) > 0 Then
                        GetTitleText = fallback
                        Exit Function
                    End If
                End If
            End If
        Next shp
    End If
End Function

' ==================================================
' DetectLanguage — same logic as main macro
' ==================================================
Private Function DetectLanguage(ByVal text As String) As String
    If Len(Trim(text)) = 0 Then
        DetectLanguage = ""
        Exit Function
    End If

    Dim i As Long, c As String
    Dim cjkCount As Long, latinCount As Long, total As Long, code As Long

    For i = 1 To Len(text)
        c = Mid(text, i, 1)
        If c <> " " And c <> vbCr And c <> vbLf And c <> vbTab Then
            code = AscW(c)
            If (code >= 65 And code <= 90) Or (code >= 97 And code <= 122) Then
                latinCount = latinCount + 1
            Else
                cjkCount = cjkCount + 1
            End If
            total = total + 1
        End If
    Next i

    If total = 0 Then
        DetectLanguage = ""
    ElseIf (cjkCount / total) >= 0.5 Then
        DetectLanguage = "ZH"
    ElseIf (latinCount / total) >= 0.5 Then
        DetectLanguage = "EN"
    Else
        DetectLanguage = "MIXED"
    End If
End Function

' ==================================================
' GetNotesText — Placeholders(2) only
' ==================================================
Private Function GetNotesText_Repro(ByRef sld As Object) As String
    Dim np As Object
    On Error Resume Next
    Set np = CallByName(sld, "NotesPage", VbGet)
    If Not np Is Nothing Then
        Dim ph As Object
        Set ph = np.Shapes.Placeholders(2)
        If Not ph Is Nothing Then
            If ph.HasTextFrame Then
                If ph.TextFrame.HasText Then
                    Dim txt As String
                    txt = Trim(ph.TextFrame.TextRange.Text)
                    If Len(txt) > 0 Then
                        GetNotesText_Repro = txt
                    End If
                End If
            End If
        End If
    End If
    On Error GoTo 0
End Function
